```
def compress_rle(image_data):
    compressed = []
    count = 1
    for i in range(1, len(image_data)):
        if image_data[i] == image_data[i-1] and count < 255:
            count += 1
        else:
            compressed.append((image_data[i-1], count))
            count = 1
    compressed.append((image_data[-1], count))
    return compressed
```

# Exemple d'utilisation
```
image = [0x00, 0x00, 0x00, 0xFF, 0xFF, 0x00]
compressed = compress_rle(image)
print(compressed)  # Output: [(0, 3), (255, 2), (0, 1)]

Décompression arduino :
```
void decompress_rle(const uint8_t *compressed, uint8_t *decompressed, size_t decompressed_size) {
  size_t index = 0;
  while (index < decompressed_size) {
    uint8_t value = pgm_read_byte(&compressed[0]);
    uint8_t count = pgm_read_byte(&compressed[1]);
    for (uint8_t i = 0; i < count; i++) {
      decompressed[index++] = value;
    }
    compressed += 2;
  }
}
